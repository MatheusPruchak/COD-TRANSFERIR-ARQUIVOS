var ABA_ASO = "JUNHO2026";

var ORIGENS = [
  { id: "1WDzmxaMd04uucyubDZqQgp0oAowKG8ow410bnWwC65g", aba: "SEED 5.4.2 - 5.4.6" }, // DV
  { id: "1nZYw-vs0ulLfaqexlO2Ohqh9bLzv761YXVTvwIf8HvA", aba: "APUCARANA" }, // AP
  { id: "1x6x9YZ6jOthcCdOlPwB7lLxFY4xX8W1Wq_LJ4AM4icQ", aba: "SEED - 5.4.1 - 5.4.7" }, // CP
  { id: "12y6TkffWHNFynSxViiKxHwtGSLQnxuzYqTN6IHq9RuY", aba: "FRANCISCO BELTRÃO" } // FB
];

function criarGatilho() {
  ScriptApp.newTrigger("aoEditar")
    .forSpreadsheet(SpreadsheetApp.getActive())
    .onEdit()
    .create();
}

function aoEditar(e) {
  if (!e) return;

  var aba = e.range.getSheet();

  if (aba.getName() !== ABA_ASO) return;

  var linha = e.range.getRow();
  var coluna = e.range.getColumn();

  if (linha < 5) return;

  // Nome digitado na coluna D
  if (coluna === 4) {
    buscarFuncionario(aba, linha);
    atualizarPagamento(aba, linha);
  }

  // Se alterar o local na coluna J, atualiza pagamento
  if (coluna === 10) {
    atualizarPagamento(aba, linha);
  }
}

function buscarFuncionario(abaDestino, linhaDestino) {
  var nomeBusca = normalizar(
    abaDestino.getRange(linhaDestino, 4).getValue()
  );

  if (nomeBusca === "") {

  abaDestino.getRange(linhaDestino, 3).clearContent();  // C Matrícula
  abaDestino.getRange(linhaDestino, 5).clearContent();  // E Telefone
  abaDestino.getRange(linhaDestino, 7).clearContent();  // G CPF
  abaDestino.getRange(linhaDestino, 8).clearContent();  // H Cargo
  abaDestino.getRange(linhaDestino, 9).clearContent();  // I Admissão
  abaDestino.getRange(linhaDestino, 10).clearContent(); // J Local
  abaDestino.getRange(linhaDestino, 15).clearContent(); // O Pagamento
  abaDestino.getRange(linhaDestino, 16).clearContent(); // P REALIZAR

  return;
}

  if (nomeBusca === "") return;

  for (var p = 0; p < ORIGENS.length; p++) {

    var planilha = SpreadsheetApp.openById(ORIGENS[p].id);
    var abaOrigem = planilha.getSheetByName(ORIGENS[p].aba);

    if (!abaOrigem) continue;

    var dados = abaOrigem.getDataRange().getDisplayValues();

    for (var i = 1; i < dados.length; i++) {

      var nomeLinha = normalizar(dados[i][5]); // Nome F

      if (nomeLinha === nomeBusca) {

        var matricula = dados[i][4];  // E
        var local = dados[i][2];      // C
        var cargo = dados[i][6];      // G
        var cpf = dados[i][7];        // H
        var telefone = dados[i][8];   // I
        var admissao = dados[i][9];   // J

        abaDestino.getRange(linhaDestino, 3).setValue(matricula); // C
        abaDestino.getRange(linhaDestino, 5).setValue(telefone); // E
        abaDestino.getRange(linhaDestino, 7).setValue(cpf);      // G
        abaDestino.getRange(linhaDestino, 8).setValue(converterCargo(cargo)); // H
        abaDestino.getRange(linhaDestino, 9).setValue(admissao); // I
        abaDestino.getRange(linhaDestino, 10).setValue(converterLocal(local)); // J
        abaDestino.getRange(linhaDestino, 16).setValue("REALIZAR"); // P
        abaDestino.getRange(linhaDestino, 14).setValue("AGENDAR"); // N

        SpreadsheetApp.getActiveSpreadsheet()
          .toast("Funcionário encontrado em " + ORIGENS[p].aba);

        return;
      }
    }
  }

  SpreadsheetApp.getActiveSpreadsheet()
    .toast("Funcionário não encontrado");
}

function atualizarPagamento(aba, linha) {
  SpreadsheetApp.flush();

  var clinica = normalizar(aba.getRange(linha, 11).getValue()); // K
  var pagamento = aba.getRange(linha, 15); // O

  var clinicasFimMes = [
    "3G ESPACO SAUDE APUCARANA",
    "BITTENCOURT",
    "BRASILMED",
    "PRO SAUDE",
    "VIZIMED",
    "VIZIMED OU ESPACO SAUDE",
    "MULTICLIM",
    "ESPACO SAUDE",
    "SAUDE DO VALE",
    "MODELLI"
  ];

  if (clinica === "") return;

  if (clinicasFimMes.includes(clinica)) {
    pagamento.setValue("FIM DO MÊS");
  } else {
    pagamento.setValue("SOLICITAR");
  }
}

function converterCargo(cargo) {
  var c = normalizar(cargo);

  if (c.includes("ADM")) return "ASS. ADMINISTRATIVO";
  if (c.includes("APOIO")) return "APOIO ESCOLAR";
  if (c.includes("COPE")) return "COPEIRA";
  if (c.includes("ENC")) return "ENCARREGADO";
  if (c.includes("INSP")) return "INSPETOR DE ALUNOS";
  if (c.includes("MERE")) return "MERENDEIRA";
  if (c.includes("LIMP")) return "SERVENTE DE LIMPEZA";

  return cargo;
}

function converterLocal(local) {
  return normalizar(local);
}

function normalizar(texto) {
  return String(texto)
    .replace(/\n/g, " ")
    .replace(/\r/g, " ")
    .replace(/\s+/g, " ")
    .trim()
    .toUpperCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "");
}
